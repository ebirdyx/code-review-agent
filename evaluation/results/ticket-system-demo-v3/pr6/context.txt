from __future__ import annotations

from dataclasses import dataclass, field

from enum import StrEnum


class Status(StrEnum):
    OPEN = "open"
    IN_PROGRESS = "in_progress"
    RESOLVED = "resolved"

class Priority(StrEnum):
    LOW = "low"
    MEDIUM = "medium"
    HIGH = "high"
    URGENT = "urgent"

    id: int
    title: str

    status: Status = Status.OPEN
    assignee: str | None = None
    created_at: datetime = field(default_factory=lambda: datetime.now(UTC))
    due_date: date | None = None

"""Business operations on tickets."""

from __future__ import annotations

from datetime import date

from .models import Priority, Status, Ticket
from .store import TicketStore
from .workflow import transition

MAX_TITLE = 120


class TicketNotFound(Exception):
    pass


class TicketService:
    def __init__(self, store: TicketStore):
        self.store = store

    def create_ticket(self, title: str, description: str, reporter: str, priority: Priority = Priority.MEDIUM) -> Ticket:
        """Create an open ticket. The title is required and at most MAX_TITLE characters."""
        title = title.strip()
        if not title:
            raise ValueError("a ticket needs a title")
        if len(title) > MAX_TITLE:
            raise ValueError(f"the title is longer than {MAX_TITLE} characters")
        return self.store.add(Ticket(id=0, title=title, description=description, reporter=reporter, priority=priority))

    def get(self, ticket_id: int) -> Ticket:
        ticket = self.store.get(ticket_id)
        if ticket is None:
            raise TicketNotFound(ticket_id)
        return ticket

    def assign(self, ticket_id: int, assignee: str) -> Ticket:
        ticket = self.get(ticket_id)
        ticket.assignee = assignee
        self.store.save(ticket)
        return ticket

    def set_due_date(self, ticket_id: int, due: date | None) -> Ticket:
        """Set or clear the due date of a ticket."""
        ticket = self.get(ticket_id)
        ticket.due_date = due
        self.store.save(ticket)
        return ticket

    def overdue(self, today: date | None = None) -> list[Ticket]:
        """Tickets whose due date has passed and that are not closed."""
        today = today or date.today()
        return [
            t for status in (Status.OPEN, Status.IN_PROGRESS, Status.RESOLVED)
            for t in self.store.list_by_status(status)
            if t.due_date is not None and t.due_date < today
        ]

    def change_status(self, ticket_id: int, new_status: Status) -> Ticket:
        ticket = self.get(ticket_id)
        transition(ticket, new_status)
        self.store.save(ticket)
        return ticket

"""SQLite storage for tickets. All SQL uses bound parameters."""

from __future__ import annotations

import sqlite3
from datetime import date, datetime

from .models import Priority, Status, Ticket

SCHEMA = """
CREATE TABLE IF NOT EXISTS tickets (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,
    description TEXT NOT NULL,
    reporter TEXT NOT NULL,
    priority TEXT NOT NULL,
    status TEXT NOT NULL,
    assignee TEXT,
    created_at TEXT NOT NULL,
    due_date TEXT
)
"""


class TicketStore:
    def __init__(self, path: str = ":memory:"):
        self._db = sqlite3.connect(path)
        self._db.row_factory = sqlite3.Row
        self._db.execute(SCHEMA)

    def add(self, ticket: Ticket) -> Ticket:
        """Insert a ticket and return it with its new id."""
        cursor = self._db.execute(
            "INSERT INTO tickets (title, description, reporter, priority, status, assignee, created_at, due_date)"
            " VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
            (ticket.title, ticket.description, ticket.reporter, ticket.priority.value, ticket.status.value,
             ticket.assignee, ticket.created_at.isoformat(), ticket.due_date.isoformat() if ticket.due_date else None),
        )
        self._db.commit()
        ticket.id = cursor.lastrowid
        return ticket

    def get(self, ticket_id: int) -> Ticket | None:
        row = self._db.execute("SELECT * FROM tickets WHERE id = ?", (ticket_id,)).fetchone()
        return _from_row(row) if row else None

    def save(self, ticket: Ticket) -> None:
        self._db.execute(
            "UPDATE tickets SET title = ?, description = ?, priority = ?, status = ?, assignee = ?, due_date = ?"
            " WHERE id = ?",
            (ticket.title, ticket.description, ticket.priority.value, ticket.status.value, ticket.assignee,
             ticket.due_date.isoformat() if ticket.due_date else None, ticket.id),
        )
        self._db.commit()

    def list_by_status(self, status: Status) -> list[Ticket]:
        rows = self._db.execute("SELECT * FROM tickets WHERE status = ? ORDER BY id", (status.value,)).fetchall()
        return [_from_row(r) for r in rows]

    def search_title(self, term: str) -> list[Ticket]:
        rows = self._db.execute(
            "SELECT * FROM tickets WHERE title LIKE ? ORDER BY id", (f"%{term}%",)
        ).fetchall()
        return [_from_row(r) for r in rows]


def _from_row(row: sqlite3.Row) -> Ticket:
    return Ticket(
        id=row["id"], title=row["title"], description=row["description"], reporter=row["reporter"],
        priority=Priority(row["priority"]), status=Status(row["status"]), assignee=row["assignee"],
        created_at=datetime.fromisoformat(row["created_at"]),
        due_date=date.fromisoformat(row["due_date"]) if row["due_date"] else None,
    )

"""Business operations on tickets."""

from __future__ import annotations

from datetime import date

from .models import Priority, Status, Ticket
from .store import TicketStore
from .workflow import transition

MAX_TITLE = 120


class TicketNotFound(Exception):
    pass


class TicketService:
    def __init__(self, store: TicketStore):
        self.store = store

    def create_ticket(self, title: str, description: str, reporter: str, priority: Priority = Priority.MEDIUM) -> Ticket:
        """Create an open ticket. The title is required and at most MAX_TITLE characters."""
        title = title.strip()
        if not title:
            raise ValueError("a ticket needs a title")
        if len(title) > MAX_TITLE:
            raise ValueError(f"the title is longer than {MAX_TITLE} characters")
        return self.store.add(Ticket(id=0, title=title, description=description, reporter=reporter, priority=priority))

    def get(self, ticket_id: int) -> Ticket:
        ticket = self.store.get(ticket_id)
        if ticket is None:
            raise TicketNotFound(ticket_id)
        return ticket

    def assign(self, ticket_id: int, assignee: str) -> Ticket:
        ticket = self.get(ticket_id)
        ticket.assignee = assignee
        self.store.save(ticket)
        return ticket

    def set_due_date(self, ticket_id: int, due: date | None) -> Ticket:
        """Set or clear the due date of a ticket."""
        ticket = self.get(ticket_id)
        ticket.due_date = due
        self.store.save(ticket)
        return ticket

    def overdue(self, today: date | None = None) -> list[Ticket]:
        """Tickets whose due date has passed and that are not closed."""
        today = today or date.today()
        return [
            t for status in (Status.OPEN, Status.IN_PROGRESS, Status.RESOLVED)
            for t in self.store.list_by_status(status)
            if t.due_date is not None and t.due_date < today
        ]

    def change_status(self, ticket_id: int, new_status: Status) -> Ticket:
        ticket = self.get(ticket_id)
        transition(ticket, new_status)
        self.store.save(ticket)
        return ticket

"""SQLite storage for tickets. All SQL uses bound parameters."""

from __future__ import annotations

import sqlite3
from datetime import date, datetime

from .models import Priority, Status, Ticket

SCHEMA = """
CREATE TABLE IF NOT EXISTS tickets (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,
    description TEXT NOT NULL,
    reporter TEXT NOT NULL,
    priority TEXT NOT NULL,
    status TEXT NOT NULL,
    assignee TEXT,
    created_at TEXT NOT NULL,
    due_date TEXT
)
"""


class TicketStore:
    def __init__(self, path: str = ":memory:"):
        self._db = sqlite3.connect(path)
        self._db.row_factory = sqlite3.Row
        self._db.execute(SCHEMA)

    def add(self, ticket: Ticket) -> Ticket:
        """Insert a ticket and return it with its new id."""
        cursor = self._db.execute(
            "INSERT INTO tickets (title, description, reporter, priority, status, assignee, created_at, due_date)"
            " VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
            (ticket.title, ticket.description, ticket.reporter, ticket.priority.value, ticket.status.value,
             ticket.assignee, ticket.created_at.isoformat(), ticket.due_date.isoformat() if ticket.due_date else None),
        )
        self._db.commit()
        ticket.id = cursor.lastrowid
        return ticket

    def get(self, ticket_id: int) -> Ticket | None:
        row = self._db.execute("SELECT * FROM tickets WHERE id = ?", (ticket_id,)).fetchone()
        return _from_row(row) if row else None

    def save(self, ticket: Ticket) -> None:
        self._db.execute(
            "UPDATE tickets SET title = ?, description = ?, priority = ?, status = ?, assignee = ?, due_date = ?"
            " WHERE id = ?",
            (ticket.title, ticket.description, ticket.priority.value, ticket.status.value, ticket.assignee,
             ticket.due_date.isoformat() if ticket.due_date else None, ticket.id),
        )
        self._db.commit()

    def list_by_status(self, status: Status) -> list[Ticket]:
        rows = self._db.execute("SELECT * FROM tickets WHERE status = ? ORDER BY id", (status.value,)).fetchall()
        return [_from_row(r) for r in rows]

    def search_title(self, term: str) -> list[Ticket]:
        rows = self._db.execute(
            "SELECT * FROM tickets WHERE title LIKE ? ORDER BY id", (f"%{term}%",)
        ).fetchall()
        return [_from_row(r) for r in rows]


def _from_row(row: sqlite3.Row) -> Ticket:
    return Ticket(
        id=row["id"], title=row["title"], description=row["description"], reporter=row["reporter"],
        priority=Priority(row["priority"]), status=Status(row["status"]), assignee=row["assignee"],
        created_at=datetime.fromisoformat(row["created_at"]),
        due_date=date.fromisoformat(row["due_date"]) if row["due_date"] else None,
    )
