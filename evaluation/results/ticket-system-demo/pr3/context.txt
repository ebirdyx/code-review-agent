"""Business operations on tickets."""

from __future__ import annotations

from .models import Priority, Status, Ticket
from .store import TicketStore
from .workflow import transition

MAX_TITLE = 120


class TicketNotFound(Exception):
    pass


class TicketService:
    def __init__(self, store: TicketStore):
        self.store = store

    def create_ticket(self, title: str, description: str, reporter: str, priority: Priority = Priority.MEDIUM) -> Ticket:
        """Create an open ticket. The title is required and at most MAX_TITLE characters."""
        title = title.strip()
        if not title:
            raise ValueError("a ticket needs a title")
        if len(title) > MAX_TITLE:
            raise ValueError(f"the title is longer than {MAX_TITLE} characters")
        return self.store.add(Ticket(id=0, title=title, description=description, reporter=reporter, priority=priority))

    def get(self, ticket_id: int) -> Ticket:
        ticket = self.store.get(ticket_id)
        if ticket is None:
            raise TicketNotFound(ticket_id)
        return ticket

    def assign(self, ticket_id: int, assignee: str) -> Ticket:
        ticket = self.get(ticket_id)
        ticket.assignee = assignee
        self.store.save(ticket)
        return ticket

    def find_by_assignee(self, name: str) -> list[Ticket]:
        """Tickets assigned to `name` (case does not matter); empty if nobody matches."""
        return self.store.search_assignee(name.strip())

    def change_status(self, ticket_id: int, new_status: Status) -> Ticket:
        ticket = self.get(ticket_id)
        transition(ticket, new_status)
        self.store.save(ticket)
        return ticket

"""SQLite storage for tickets. All SQL uses bound parameters."""

from __future__ import annotations

import sqlite3
from datetime import datetime

from .models import Priority, Status, Ticket

SCHEMA = """
CREATE TABLE IF NOT EXISTS tickets (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,
    description TEXT NOT NULL,
    reporter TEXT NOT NULL,
    priority TEXT NOT NULL,
    status TEXT NOT NULL,
    assignee TEXT,
    created_at TEXT NOT NULL
)
"""


class TicketStore:
    def __init__(self, path: str = ":memory:"):
        self._db = sqlite3.connect(path)
        self._db.row_factory = sqlite3.Row
        self._db.execute(SCHEMA)

    def add(self, ticket: Ticket) -> Ticket:
        """Insert a ticket and return it with its new id."""
        cursor = self._db.execute(
            "INSERT INTO tickets (title, description, reporter, priority, status, assignee, created_at)"
            " VALUES (?, ?, ?, ?, ?, ?, ?)",
            (ticket.title, ticket.description, ticket.reporter, ticket.priority.value, ticket.status.value,
             ticket.assignee, ticket.created_at.isoformat()),
        )
        self._db.commit()
        ticket.id = cursor.lastrowid
        return ticket

    def get(self, ticket_id: int) -> Ticket | None:
        row = self._db.execute("SELECT * FROM tickets WHERE id = ?", (ticket_id,)).fetchone()
        return _from_row(row) if row else None

    def save(self, ticket: Ticket) -> None:
        self._db.execute(
            "UPDATE tickets SET title = ?, description = ?, priority = ?, status = ?, assignee = ? WHERE id = ?",
            (ticket.title, ticket.description, ticket.priority.value, ticket.status.value, ticket.assignee, ticket.id),
        )
        self._db.commit()

    def list_by_status(self, status: Status) -> list[Ticket]:
        rows = self._db.execute("SELECT * FROM tickets WHERE status = ? ORDER BY id", (status.value,)).fetchall()
        return [_from_row(r) for r in rows]

    def search_title(self, term: str) -> list[Ticket]:
        rows = self._db.execute(
            "SELECT * FROM tickets WHERE title LIKE ? ORDER BY id", (f"%{term}%",)
        ).fetchall()
        return [_from_row(r) for r in rows]


    def search_assignee(self, name: str) -> list[Ticket]:
        query = f"SELECT * FROM tickets WHERE LOWER(assignee) = '{name.lower()}' ORDER BY id"
        rows = self._db.execute(query).fetchall()
        return [_from_row(r) for r in rows]


def _from_row(row: sqlite3.Row) -> Ticket:
    return Ticket(
        id=row["id"], title=row["title"], description=row["description"], reporter=row["reporter"],
        priority=Priority(row["priority"]), status=Status(row["status"]), assignee=row["assignee"],
        created_at=datetime.fromisoformat(row["created_at"]),
    )

"""Business operations on tickets."""

from __future__ import annotations

from .models import Priority, Status, Ticket
from .store import TicketStore
from .workflow import transition

MAX_TITLE = 120


class TicketNotFound(Exception):
    pass


class TicketService:
    def __init__(self, store: TicketStore):
        self.store = store

    def create_ticket(self, title: str, description: str, reporter: str, priority: Priority = Priority.MEDIUM) -> Ticket:
        """Create an open ticket. The title is required and at most MAX_TITLE characters."""
        title = title.strip()
        if not title:
            raise ValueError("a ticket needs a title")
        if len(title) > MAX_TITLE:
            raise ValueError(f"the title is longer than {MAX_TITLE} characters")
        return self.store.add(Ticket(id=0, title=title, description=description, reporter=reporter, priority=priority))

    def get(self, ticket_id: int) -> Ticket:
        ticket = self.store.get(ticket_id)
        if ticket is None:
            raise TicketNotFound(ticket_id)
        return ticket

    def assign(self, ticket_id: int, assignee: str) -> Ticket:
        ticket = self.get(ticket_id)
        ticket.assignee = assignee
        self.store.save(ticket)

"""SQLite storage for tickets. All SQL uses bound parameters."""

from __future__ import annotations

import sqlite3
from datetime import datetime

from .models import Priority, Status, Ticket

SCHEMA = """
CREATE TABLE IF NOT EXISTS tickets (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,
    description TEXT NOT NULL,
    reporter TEXT NOT NULL,
    priority TEXT NOT NULL,
    status TEXT NOT NULL,
    assignee TEXT,
    created_at TEXT NOT NULL
)
"""


class TicketStore:
    def __init__(self, path: str = ":memory:"):
        self._db = sqlite3.connect(path)
        self._db.row_factory = sqlite3.Row
        self._db.execute(SCHEMA)

    def add(self, ticket: Ticket) -> Ticket:
        """Insert a ticket and return it with its new id."""
        cursor = self._db.execute(
            "INSERT INTO tickets (title, description, reporter, priority, status, assignee, created_at)"
            " VALUES (?, ?, ?, ?, ?, ?, ?)",
            (ticket.title, ticket.description, ticket.reporter, ticket.priority.value, ticket.status.value,
             ticket.assignee, ticket.created_at.isoformat()),
        )
        self._db.commit()
        ticket.id = cursor.lastrowid
        return ticket

    def get(self, ticket_id: int) -> Ticket | None:
        row = self._db.execute("SELECT * FROM tickets WHERE id = ?", (ticket_id,)).fetchone()
        return _from_row(row) if row else None

    def save(self, ticket: Ticket) -> None:
        self._db.execute(
            "UPDATE tickets SET title = ?, description = ?, priority = ?, status = ?, assignee = ? WHERE id = ?",
            (ticket.title, ticket.description, ticket.priority.value, ticket.status.value, ticket.assignee, ticket.id),
        )
        self._db.commit()

    def list_by_status(self, status: Status) -> list[Ticket]:
        rows = self._db.execute("SELECT * FROM tickets WHERE status = ? ORDER BY id", (status.value,)).fetchall()
        return [_from_row(r) for r in rows]

    def search_title(self, term: str) -> list[Ticket]:
        rows = self._db.execute(
            "SELECT * FROM tickets WHERE title LIKE ? ORDER BY id", (f"%{term}%",)
        ).fetchall()
        return [_from_row(r) for r in rows]


    def search_assignee(self, name: str) -> list[Ticket]:
        query = f"SELECT * FROM tickets WHERE LOWER(assignee) = '{name.lower()}' ORDER BY id"
        rows = self._db.execute(query).fetchall()
        return [_from_row(r) for r in rows]


def _from_row(row: sqlite3.Row) -> Ticket:
    return Ticket(
        id=row["id"], title=row["title"], description=row["description"], reporter=row["reporter"],
        priority=Priority(row["priority"]), status=Status(row["status"]), assignee=row["assignee"],
        created_at=datetime.fromisoformat(row["created_at"]),
    )
